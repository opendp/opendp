\subsection*{Preconditions}
None.

\subsection*{Postcondition}
\begin{theorem}
    \validTransformation{\texttt{(input\_domain, input\_metric)}}{\\ \texttt{make\_np\_sum}}
\end{theorem}

\begin{proof}
Let $x,x'$ be members of \texttt{input\_domain}.
Let $R=\texttt{desc.norm}$, $O=\texttt{desc.origin}$,
$p=\texttt{desc.p}\in\{1,2\}$, and $n$ be the maximum row count used by the constructor.
Under idealized domain membership, for each row $v$,
\[
    \norm{v-O}_p\le R,\qquad
    \norm{v}_p\le \norm{v-O}_p+\norm{O}_p\le R+\norm{O}_p\le B,
\]
where $B=\texttt{row\_bound}$.

\paragraph{Exact sensitivity.}
Let $s(x)=\sum_i x_i$.
Let $M,M'$ be the multisets of rows in $x,x'$,
$C=M\cap M'$, $A=M\setminus C$, and $A'=M'\setminus C$.
Then
\begin{align*}
    s(x)-s(x') &= \sum_{a\in A}a-\sum_{b\in A'}b,\\
    |A|+|A'| &= d_{\mathrm{Sym}}(x,x')\le \din,\\
    |A|,|A'| &\le n.
\end{align*}

If \texttt{desc.size} is known, $m=|A|=|A'|\le\min(\lfloor\din/2\rfloor,n)$.
Enumerate $A=\{a_1,\ldots,a_m\}$ and $A'=\{b_1,\ldots,b_m\}$.
\begin{align*}
    \norm{s(x)-s(x')}_p
    &= \norm{\sum_{i=1}^m(a_i-b_i)}_p\\
    &\le \sum_{i=1}^m\norm{a_i-b_i}_p\\
    &\le \sum_{i=1}^m
        \left(\norm{a_i-O}_p+\norm{b_i-O}_p\right)\\
    &\le 2mR\\
    &\le 2\min(\lfloor\din/2\rfloor,n)R.
\end{align*}

Otherwise,
\begin{align*}
    \norm{s(x)-s(x')}_p
    &= \norm{\sum_{a\in A}a-\sum_{b\in A'}b}_p\\
    &\le \sum_{a\in A}\norm{a}_p+\sum_{b\in A'}\norm{b}_p\\
    &\le (|A|+|A'|)B\\
    &\le \min(\din,2n)B.
\end{align*}
Thus $\norm{s(x)-s(x')}_p\le S(\din)$, where
\[
    S(d)=
    \begin{cases}
        2\min(\lfloor d/2\rfloor,n)R & \text{if size is known},\\
        \min(d,2n)B & \text{otherwise}.
    \end{cases}
\]

\paragraph{Floating-point summation.}
The cast from \texttt{f32} or \texttt{f64} to \texttt{f64} is exact.
For $a(x)=\sum_i|x_i|$ (coordinatewise absolute values),
\begin{align*}
    \norm{a(x)}_\infty
    &\le \norm{a(x)}_p\\
    &\le \sum_i\norm{x_i}_p\\
    &\le nB.
\end{align*}
The same bound holds for $x'$.
Each level of \texttt{\_pairwise\_sum} reduces $k$ rows to $\lceil k/2\rceil$ rows.
Hence its summation depth is at most
\[
    h=\lceil\log_2\max(1,n)\rceil.
\]
Let $U=2^{-53}$ and $H=\texttt{magnitude}\ge nB$.
By the postcondition of \texttt{sum\_error}, the constructor's public overflow check ensures finite intermediate sums, with
\[
    hU<1,\qquad
    E=\frac{hU}{1-hU}H,\qquad
    \norm{\function(x)-s(x)}_p\le E.
\]
Therefore,
\begin{align*}
    \norm{\function(x)-\function(x')}_p
    &\le \norm{\function(x)-s(x)}_p
        +\norm{s(x)-s(x')}_p
        +\norm{s(x')-\function(x')}_p\\
    &\le S(\din)+2E\\
    &\le \texttt{float\_sum\_sensitivity}
        \left(\texttt{\_round\_up}(S(\din)),h,H,H\right)\\
    &=\dout.
\end{align*}
This includes $\din=0$, when the summation orders can differ.

\paragraph{Integer summation.}
Let $L$ be the maximum representable value of the promoted output type.
The constructor requires $2nB\le L$.
For any intermediate sum $t$ over a subset $I$ of rows,
\[
    \norm{t}_\infty
    =\norm{\sum_{i\in I}x_i}_\infty
    \le \sum_{i\in I}\norm{x_i}_p
    \le nB\le L.
\]
For unsigned input, $t$ is also nonnegative coordinatewise.
Thus every integer sum is representable and $\function(x)=s(x)$.
Since $S(\din)\le 2nB\le L$,
\[
    \norm{\function(x)-\function(x')}_p
    \le S(\din)\le \lceil S(\din)\rceil=\dout\le L.
\]

In both cases, the output is a vector of the declared atomic type, without NaNs or overflow.
\end{proof}
