\subsection*{Postcondition}

\begin{theorem}
    Define \function as follows:
    \begin{equation*}
        \function(x) = \mathrm{score\_candidates}_i(x, \texttt{candidates}, \texttt{alpha\_num}, \texttt{alpha\_den}, \texttt{size\_limit})
    \end{equation*}

    The function calls \rustdoc{transformations/quantile\_score\_candidates/fn}{score\_candidates} with fixed choices of
    \texttt{candidates}, \texttt{alpha\_num}, \texttt{alpha\_den} and \texttt{size\_limit}.

    If the input domain is the set of all vectors with non-null elements of type \texttt{TI},
    input metric is either \rustdoc{metrics/struct}{SymmetricDistance} or \rustdoc{metrics/struct}{InsertDeleteDistance},
    and the postcondition of \function is satisfied, then \function is stable.

    This means that, for any two elements $x, x'$ in the \texttt{input\_domain},
        where $x, x'$ share the same length if \texttt{known\_length} is true,
    and any pair $(\din, \dout)$,
        where \din\ has the associated type for \texttt{input\_metric} and \dout\ has the associated type for \texttt{output\_metric},
    then when $x, x'$ are \din-close under \texttt{input\_metric} and $\texttt{stability\_map}(\din) \leq \dout$,
    \function(x), \function(x') are \dout-close under \texttt{output\_metric}.
\end{theorem}


The sensitivity of this function differs depending on whether the size of the input vector is known.
\label{unsized-stability}
First, consider the case where the size is unknown.

\begin{lemma}
    \label{single-change-stab-sym}
    If $d_{Sym}(x, x') = 1$, then $d_\infty(\function(x), \function(x')) \le \max(\alpha_{num}, \alpha_{den} - \alpha_{num})$.
\end{lemma}

\begin{proof}
    For each candidate $C_i$, let $a(x)=\min(\#(x<C_i),l)$ and $b(x)=\min(\#(x>C_i),l)$.
    Write $A=\alpha_{den}-\alpha_{num}$ and $B=\alpha_{num}$, both nonnegative.
    By the postcondition of \texttt{score\_candidates}, the score is $|Aa(x)-Bb(x)|$.
    The reverse triangle inequality gives
    \[
        |\function(x)_i-\function(x')_i|
        \leq A|a(x)-a(x')|+B|b(x)-b(x')|.
    \]
    A single insertion or deletion changes at most one of the two counts by one.
    Clipping at $l$ cannot increase that change, since $t\mapsto\min(t,l)$ is 1-Lipschitz.
    Thus the right-hand side is at most $\max(A,B)$.
    Taking the maximum over candidates proves the lemma.
\end{proof}

\label{sec:known-size}
Now consider the case where the dataset size is known.

\begin{lemma}
    \label{single-change-stab}
    If $d_{CO}(x, x') \le 1$, then $d_\infty(\function(x), \function(x')) \le \alpha_{den}$.
\end{lemma}

\begin{proof}
    Use the same capped counts and coefficients as in the previous lemma.
    A single replacement changes each count by at most one, even after clipping.
    The reverse triangle inequality therefore bounds each score change by
    \[
        A|a(x)-a(x')|+B|b(x)-b(x')|
        \leq A+B=\alpha_{den}.
    \]
    Taking the maximum over candidates proves the lemma.
\end{proof}

\begin{proof}[Proof of postcondition]
    Fix two inputs at distance at most \din\ under either input metric.
    If their sizes are unknown, they can be related by at most \din\ insertions and deletions (up to permutation for symmetric distance).
    Scores are invariant to permutation, since they depend only on the histogram counts.
    By the triangle inequality and Lemma~\ref{single-change-stab-sym}, the output distance is at most
    \[
        \din\cdot\max(\alpha_{num},\alpha_{den}-\alpha_{num}).
    \]

    If the inputs have equal length, an edit sequence can be chosen with equal numbers of insertions and deletions.
    After matching the common records, pair each deleted record with an inserted record.
    Up to permutation, at most $\lfloor\din/2\rfloor$ replacements suffice.
    Since the scorer is permutation-invariant, Lemma~\ref{single-change-stab} and the triangle inequality give output distance at most
    \[
        \lfloor\din/2\rfloor\cdot\alpha_{den}.
    \]
    These bounds are the respective values returned by the stability map when its checked arithmetic succeeds.
\end{proof}
