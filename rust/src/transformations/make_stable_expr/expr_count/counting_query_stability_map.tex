\subsubsection*{Caller-verified}
Each partition statistic $c$ is a count with
$|c(x)-c(x')|\le d_M(x,x')$.
If \texttt{invariant} is \texttt{Lengths}, $c$ counts all rows and
partition lengths are invariant.

\subsection*{Postcondition}
\begin{theorem}
    The returned map either errors or, for all
    $(l_0,l_1,l_\infty)$-close inputs under \texttt{L01InfDistance<M>},
    bounds $\|f(x)-f(x')\|_P$, where $f(x)_i=c(x_i)$.
\end{theorem}

\begin{proof}
    For invariant lengths, $f(x)=f(x')$.
    Otherwise, set $v_i=c(x_i)-c(x'_i)$ and $d_i=d_M(x_i,x'_i)$.
    By the input metric and the caller-verified precondition,
    \begin{align*}
        |v_i|&\le d_i,\\
        \|v\|_0&\le\|d\|_0\le l_0,\\
        \|v\|_1&\le\|d\|_1\le l_1,\\
        \|v\|_\infty&\le\|d\|_\infty\le l_\infty.
    \end{align*}
    The postcondition of \rustdoc{metrics/fn}{l01inf\_to\_lp}
    gives $\|v\|_P\le\texttt{out}$ on success.
\end{proof}
