\subsection*{Postcondition}
\begin{theorem}
    The implementation is consistent with the associated items in the \rustdoc{measurements/noisy\_top\_k/trait}{TopKMeasure} trait.
    \begin{enumerate}
        \item Method \texttt{privacy\_map}:
        For any finite score vectors $x, x'$ of the same nonzero length where $d_\mathrm{in} \ge d_\mathrm{Range}(x, x')$,
        return an error or $d_\mathrm{out} \ge D_\mathrm{self}(f(x), f(x'))$,
        where $f(x) = \mathrm{noisy\_top\_k}(x=x, k=1, \mathrm{scale}=\mathrm{scale}, \mathrm{replacement}=\mathrm{Self::REPLACEMENT})$.
    \end{enumerate}
\end{theorem}

\begin{proof}
    Since \texttt{Self::REPLACEMENT} is false, then by the postcondition of \texttt{noisy\_top\_k},
    \texttt{noisy\_top\_k} returns a sample from $\mathcal{M}_{PF}$.
    The selection distribution is invariant under adding a constant to every score, and
    \[
        d_{\mathrm{Range}}(x,x')=2\min_{c\in\mathbb R}\|x-x'-c\mathbf{1}\|_\infty.
    \]
    By Theorem~1 of \cite{mckenna2020permute}, permute-and-flip with scale $s$
    and score sensitivity $\Delta$ satisfies $2\Delta/s$-DP.
    Taking $\Delta=d_{\mathrm{Range}}(x,x')/2$ yields the bound
    $d_{\mathrm{Range}}(x,x')/s\leq d_{\mathrm{in}}/s$.
    The directed division in \texttt{privacy\_map} upper bounds this quantity.
\end{proof}
