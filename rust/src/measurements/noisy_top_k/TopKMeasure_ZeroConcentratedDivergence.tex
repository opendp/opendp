\subsection*{Postcondition}
\begin{theorem}
    The implementation is consistent with all associated items in the \rustdoc{measurements/noisy\_top\_k/trait}{TopKMeasure} trait.
    \begin{enumerate}
        \item Method \texttt{privacy\_map}:
        For any finite score vectors $x, x'$ of the same nonzero length where $d_\mathrm{in} \ge d_\mathrm{Range}(x, x')$,
        return an error or $d_\mathrm{out} \ge D_\mathrm{self}(f(x), f(x'))$,
        where $f(x) = \mathrm{noisy\_top\_k}(x=x, k=1, \mathrm{scale}=\mathrm{scale}, \mathrm{replacement}=\mathrm{Self::REPLACEMENT})$.
    \end{enumerate}
\end{theorem}

\begin{proof}[Proof of postcondition: \texttt{privacy\_map}]

    By the postcondition of \texttt{noisy\_top\_k}, for $k=1$ the output probabilities are proportional to $\exp(x_i/\texttt{scale})$.
    The log probability ratio between corresponding outputs on $x$ and $x'$ is $(x_i-x'_i)/\texttt{scale}$ plus a constant independent of $i$.
    Its range has width at most $\epsilon=d_{\mathrm{in}}/\texttt{scale}$.
    By Lemma~3.2 of \cite{cesar2020bounding}, $\epsilon$-bounded range implies $\epsilon^2/8$-zCDP.
    The directed arithmetic in \texttt{privacy\_map} upper bounds this quantity.
\end{proof}
