\section{Introduction}
This is a randomizer in the local model of differential privacy, which is a simplification of the RAPPOR protocol~\cite{rappor}. Under local differential privacy each user is guaranteed $\varepsilon$-DP for their response. RAPPOR achieves local differentially private frequency estimation by flipping each bit of an input vector with fixed probability. Because the noise is added mechanically we can efficiently post-process a set of outputs from this randomizer with the function \rustdoc{measurements/fn}{debias\_randomized\_response\_bitvec}, which sums and normalises a vector of private outputs using the quantized flip probability to account for the bias introduced. The input to this function are elements from a \rustdoc{domains/struct}{BitVectorDomain} with fixed $\texttt{max\_weight}$ and $\texttt{size}$. The former bounds sensitivity; the latter ensures that output length does not reveal the input. we detail two use cases below to explain two possible settings of $\texttt{max\_weight}$, which we will denote $m$.

Frequency estimation is the problem of generating a histogram of the empirical frequencies of categorical elements. Any categorical domain of size $k$ can be mapped to the set $[k]=\{0,1,\ldots,k-1\}$, and we can represent any element $j\in[k]$ as the $j$'th standard basis vector so that $m=1$. In cases where the domain is extremely large or even unbounded (such as a set of strings) we can hash the domain elements using a bloom filter with $m$ hash functions. In order to decode a set of outputs generated in this fashion we must perform a regression against the bloom filter representations of a set of candidate elements, neither bloom filters nor the regression tools required for this are implemented here, but details can be found in the original RAPPOR paper. We will require that $2m+1<k$, but $m$ should be much less than $k$ in order to preserve utility. As we will prove in~\ref{thm:privacy-parameter} privacy degrades linearly with $m$.

\begin{theorem}[\cite{rappor}]
\label{thm:privacy-parameter}
	The program \rustdoc{measurements/fn}{make\_randomized\_response\_bitvec} satisfies $\varepsilon$-DP, where
	\begin{equation*}
		\varepsilon = 2m\log\left(\frac{1-q}{q}\right).
	\end{equation*}
\end{theorem}
\begin{proof}
    The checks on lines~\ref{line:max-weight} and~\ref{line:size} ensure that a successfully constructed measurement has a specified maximum weight $m$ and size $k$. By the definition of \rustdoc{domains/struct}{BitVectorDomain} membership, any two inputs $x,x'$ in this domain each contain exactly $k$ bits, with at most $m$ ones. The mechanism flips each of these $k$ bits independently and therefore produces reports in the same space $\{0,1\}^k$ for both inputs. In particular, report length cannot distinguish the inputs.
	Let $y=(y_1,\ldots,y_j,\ldots,y_k)$ be a randomised report generated by applying \rustdoc{measurements/fn}{make\_randomized\_response\_bitvec} to an input vector $x=(x_1,\ldots,x_j,\ldots,x_k)$. The output of the algorithm satisfies the following conditional probabilities,
	\begin{align}
		\mathbb{P}[y_j = 1~|~x_j=1] &= 1 - q\\
		\mathbb{P}[y_j = 1~|~x_j=0] &=q
	\end{align}
	The probability of observing any given report $Y$ is $\mathbb{P}[Y=y | X=x]$. Suppose $x=(x,\ldots,x_k)$ is a single Boolean vector with at most $m$ ones.
	Without loss of generality, as the probabilities are invariant under permutation, assume that $x^*=(x=1,\ldots,x_m=1,x_{m+1}=0,\ldots,x_k=0)$, then we have
	\begin{align*}
		\mathbb{P}[Y=y~|~X=x^*] &=%
			\prod\limits_{i=1}^m \left(q\right)^{1-y_i}\left(1-q\right)^{y_i}%
			\times \prod\limits_{i=m+1}^k\left(q\right)^{y_i} \left(1-q\right)^{1-y_i}.
	\end{align*}
	Then let $D$ be the ratio of two such conditional probabilities with neighbouring inputs $x,x'$, and let $S$ be the range of \rustdoc{measurements/fn}{make\_randomized\_response\_bitvec}.
	\begin{align}
		D &= \frac{\mathbb{P}[Y\in S~|~X=x]}{\mathbb{P}[Y\in S~|~X=x']}\\
			&= \frac{\sum_{y\in S}\mathbb{P}[Y=y~|~X=x]}{\sum_{y\in S}\mathbb{P}[Y=y~|~X=x']}\nonumber\\
			&\leq \max_{y\in S}\frac{\mathbb{P}[Y=y~|~X=x]}{\mathbb{P}[Y=y~|~X=x']}\nonumber\\
			&=\max_{y\in S}\frac{%
				\prod\limits_{j=1}^m \left(q\right)^{1-y_j}\left(1-q\right)^{y_j}%
				\times \prod\limits_{j=m+1}^k\left(q\right)^{y_j} \left(1-q\right)^{1-y_j}
			}{%
				\prod\limits_{j=1}^m \left(q\right)^{y_j}\left(1-q\right)^{1-y_j}%
				\times \prod\limits_{j=m+1}^{2m}\left(q\right)^{1-y_j} \left(1-q\right)^{y_j}\times%
				\prod\limits_{j=2m+1}^{k}\left(q\right)^{y_j}\left(1-q\right)^{1-y_j}
			}\nonumber\\
			&=\max_{y\in S}\frac{%
				\prod\limits_{j=1}^m \left(q\right)^{1-y_j}\left(1-q\right)^{y_j}%
				\times \prod\limits_{j=m+1}^{2m}\left(q\right)^{y_j} \left(1-q\right)^{1-y_j}%
			}{%
				\prod\limits_{j=1}^m \left(q\right)^{y_j}\left(1-q\right)^{1-y_j}%
				\times \prod\limits_{j=m+1}^{2m}\left(q\right)^{1-y_j} \left(1-q\right)^{y_j}%
			}\label{eq:privacy:cancel-k}\\
			&=\max_{y\in S}\left[%
				\prod\limits_{j=1}^m \left(q\right)^{2(1-y_j)}\left(1-q\right)^{2y_j}%
				\times \prod\limits_{j=m+1}^{2m}\left(q\right)^{2y_j} \left(1-q\right)^{2(1-y_j)}\right]\label{eq:maximise-product}%
	\end{align}
	Notice that, by Equation~\ref{eq:privacy:cancel-k}, the privacy is not dependent on $k$ and Equation~\ref{eq:maximise-product} is maximised when $y=1,\ldots,y_m=1$, and $y_{m+1},\ldots,y_{2m}=0$, giving
	\begin{align*}
		D &\leq \left(1-q\right)^{2m}\times\left(q\right)^{-2m}\\
			&= \left(\frac{1-q}{q}\right)^{2m}
	\end{align*}
	Therefore,
	\begin{equation}
		\varepsilon \leq 2m\log\left(\frac{1-q}{q}\right),
	\end{equation}
    which completes the proof.
\end{proof}

\subsection{Utility}

\begin{theorem}
\label{thm:unbiased-estimator}
	The program \rustdoc{measurements/fn}{debias\_randomized\_response\_bitvec}, using the effective quantized flip probability $q$, is an unbiased frequency estimator when $q<1/2$ (in exact arithmetic).
\end{theorem}
\begin{proof}
By independence we can focus on the contribution of one message. Given an input $\bar{x}=(x_1,\ldots,x_j,\ldots,x_k)$, denote the output of \rustdoc{measurements/fn}{make\_randomized\_response\_bitvec} on this input as $\bar{y}$. Assume each $x_j$ was drawn from a distribution with probability $p(x_j)$, we can replace this assumption with the true frequency of $x_j$ later. Each coordinate of $\bar{y}$ is therefore given as follows,
\begin{equation*}
    y_j = \text{Bern}\left(1-q\right)x_j + \text{Bern}\left(q\right) (1-x_j).
\end{equation*}
And the expectation of $\bar{y}_j$ is
\begin{align}
    \mathbb{E}\left[y_j\right] &= (1-q)p(x_j) + q(1-p(x_j))\nonumber\\
        &= (1-2q)p(x_j) + q\label{eq:expect-y}
\end{align}
Therefore rearranging~\ref{eq:expect-y} we can estimate $p(x_j)$ as,
\begin{equation}
    \hat{p}(x)=\frac{y_j - q}{1-2q}
\end{equation}
Given $n$ outputs, let $Y=\sum_{i=1}^n \bar{y}_i$ be the element-wise sum of the vector outputs. We can use the above to create an estimator the true counts $\hat{h}$ as,
\begin{equation*}
    \hat{h}_j = \frac{Y_j-nq}{1-2q}\numberthis\label{eq:estimator}
\end{equation*}
Which is the estimator in \rustdoc{measurements/fn}{debias\_randomized\_response\_bitvec}.
\end{proof}

\begin{theorem}
	The program \rustdoc{measurements/fn}{debias\_randomized\_response\_bitvec} is a frequency estimator with mean squared error
	\begin{equation*}
		\mathbb{E}[\|h-\hat{h}\|_2^2] = \frac{nk q(1-q)}{(1-2q)^2}.
	\end{equation*}
\end{theorem}
\begin{proof}
    Let $h = h(X^n)=\sum\limits_{j=1}^n \bar{x}_i$ be the true (non-private) vector sum of the inputs. Our goal is to find the mean squared error of our estimate from this vector. Notice that each coordinate $Y_j$ of $Y=\sum_{i=1}^n \bar{y}_i$, is a sum of Bernoulli random variables with probability $q$ or $(1-q)$ meaning their sum is that of two binomials with equal variance (by the symmetry of their probabilities). Using this we get that the variance is,
\begin{equation}
    \text{Var}(Y_j) = nq\left(1-q\right), \label{eq:varianceY}
\end{equation}
and the mean squared error is,
\begin{align*}
    \mathbb{E}[\|\hat{h}-h\|_2^2] &= \mathbb{E}\left[\sum\limits_{j=1}^k(\hat{h_j}-h_j)^2\right] = \sum\limits_{j=1}^k\mathbb{E}[(\hat{h_j}-h_j)^2] \\
			&= \sum\limits_{j=1}^k\mathbb{E}[(\hat{h_j}-\mathbb{E}[\hat{h_j}])^2] \tag{By unbiasedness, Theorem~\ref{thm:unbiased-estimator}}\\
			&= \sum\limits_{j=1}^k\text{Var}(\hat{h_j}) = \sum\limits_{j=1}^k\text{Var}\left(\frac{\hat{Y_j}-nq}{1-2q}\right) \tag{By Equation~\ref{eq:estimator}}\\
			&= \sum\limits_{j=1}^k\text{Var}\left(\frac{Y_j}{1-2q}\right) \\
			&= \sum\limits_{j=1}^k\frac{\text{Var}(Y_i)}{(1-2q)^2}\\
			&= k\left(\frac{nq\left(1-q\right)}{(1-2q)^2}\right) \tag{By Equation~\ref{eq:varianceY}}\\
			& = \frac{nk q(1-q)}{(1-2q)^2}
\end{align*}
\end{proof}
