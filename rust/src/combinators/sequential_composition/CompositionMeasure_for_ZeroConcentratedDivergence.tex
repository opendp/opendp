\begin{theorem}
  \texttt{composability} returns \texttt{Ok(out)} if the composition of a vector of privacy parameters \texttt{d\_mids}
  is bounded above by \texttt{self.compose(d\_mids)} under \texttt{adaptivity} adaptivity and \texttt{out}-composability.
  Otherwise returns an error.
\end{theorem}


\begin{lemma}
  \label{lemma:zcdp}
  \begin{enumerate}
    \item $\mathcal{F}(\rho_1, \rho_2, \ldots; \rho) = \mathbb{I}(\sum_i \rho_i \leq \rho)$ is a valid $\rho$-zCDP IM-filter.
    \item $\mathcal{G}(\rho_1, \ldots, \rho_k) = \sum_{i=1}^k \rho_i$ is a valid $\rho$-zCDP IM-privacy loss accumulator.
  \end{enumerate}
\end{lemma}

\begin{proof}
  Fix any $\alpha>1$ and set $\epsilon_i=\alpha\rho_i$ and $\epsilon=\alpha\rho$.
  By the proof definition of \rustdoc{measures/struct}{ZeroConcentratedDivergence}, each $\rho_i$-zCDP mechanism is $(\alpha,\epsilon_i)$-RDP.
  The filter condition $\sum_i\rho_i\leq\rho$ is equivalent to $\sum_i\epsilon_i\leq\epsilon$.
  By Theorem~1.22(1) of \cite{haney2025concurrent}, this is a valid interactive RDP filter.
  Since the argument holds for every $\alpha>1$, it is a valid interactive zCDP filter.
  The corresponding accumulator $\mathcal{G}$ is valid by the filter-accumulator equivalence (Lemma~2.19 of \cite{haney2025concurrent}).
\end{proof}

\begin{proof}
  By the postcondition of \rustdoc{traits/trait}{InfAdd} we have that $\sum_i \texttt{d\_mids}_i \le \texttt{compose(d\_mids)}$.

      Fully-Adaptive & Remark~4.4\cite{feldman2022individual} & \ref{lemma:zcdp}

\end{proof}
